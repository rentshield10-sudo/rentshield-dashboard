6. First Auto-Reply Rule

The first auto-reply rule is:

If any inbound reply comes from a phone number
AND exactly one matching dashboard lead exists for that phone
AND that lead has current_status = contacted
AND that lead has conversation_stage = intro_sent
AND automation is not stopped
AND human review is not already required
AND lead has not already moved to requirements_sent/booked/etc

THEN:
send 02_requirements_request_v1 template

ELSE:
send nothing
mark/send to Human Review

This handles the issue where the same phone number may inquire again weeks later for a different apartment.

Safe rule:

0 matching intro_sent/contacted leads
→ Human Review / no auto-send

1 matching intro_sent/contacted lead
→ Send requirements template

2+ matching intro_sent/contacted leads
→ Human Review / no auto-send

This avoids guessing which apartment the person means.

9. Corrected Lookup Function To Run Next

Run this next in Supabase SQL Editor:

drop function if exists public.find_single_contacted_lead_for_auto_reply(text);

create or replace function public.find_single_contacted_lead_for_auto_reply(
  p_phone text
)
returns table (
  matched_count bigint,
  lead_id text,
  lead_name text,
  phone text,
  apt_address text,
  pipeline_status text,
  current_status text,
  conversation_stage text,
  last_outbound_sms text,
  last_outbound_at text,
  stop_automation text,
  needs_human_review text
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_normalized_phone text;
begin
  v_normalized_phone := public.normalize_phone_text(p_phone);

  return query
  with matches as (
    select
      dl.lead_id::text as lead_id,
      dl.lead_name::text as lead_name,
      dl.phone::text as phone,
      dl.apt_address::text as apt_address,
      dl.pipeline_status::text as pipeline_status,
      dl.current_status::text as current_status,
      dl.conversation_stage::text as conversation_stage,
      dl.last_outbound_sms::text as last_outbound_sms,
      dl.last_outbound_at::text as last_outbound_at,
      dl.stop_automation::text as stop_automation,
      dl.needs_human_review::text as needs_human_review
    from public.dashboard_leads dl
    where public.normalize_phone_text(dl.phone) = v_normalized_phone
      and lower(coalesce(dl.current_status, '')) = 'contacted'
      and lower(coalesce(dl.conversation_stage, '')) = 'intro_sent'
      and lower(coalesce(dl.pipeline_status, 'new_lead')) not in (
        'requirements_sent',
        'schedule_sent',
        'booked',
        'confirmed',
        'showed',
        'no_show',
        'cancelled',
        'reschedule_needed',
        'lost',
        'nurture'
      )
      and lower(coalesce(dl.stop_automation::text, 'false')) not in ('true', 'yes', '1')
      and lower(coalesce(dl.needs_human_review::text, 'false')) not in ('true', 'yes', '1')
  ),
  counted as (
    select count(*) as matched_count from matches
  )
  select
    counted.matched_count,
    matches.lead_id,
    matches.lead_name,
    matches.phone,
    matches.apt_address,
    matches.pipeline_status,
    matches.current_status,
    matches.conversation_stage,
    matches.last_outbound_sms,
    matches.last_outbound_at,
    matches.stop_automation,
    matches.needs_human_review
  from counted
  left join matches on counted.matched_count = 1;
end;
$$;

grant execute on function public.find_single_contacted_lead_for_auto_reply(text) to anon;
grant execute on function public.find_single_contacted_lead_for_auto_reply(text) to authenticated;

16. QuoSender Server Logic To Add Later

When QuoSender receives an inbound message:

async function handleInboundQuoMessage(inbound) {
  const phone = inbound.phone;
  const inboundText = inbound.message;

  // 1. Ask Supabase if this phone has exactly one eligible lead.
  const lookup = await supabase.rpc(
    "find_single_contacted_lead_for_auto_reply",
    { p_phone: phone }
  );

  const row = lookup.data?.[0];

  if (!row || row.matched_count !== 1) {
    // no send
    // human review later
    return {
      action: "human_review",
      reason: `matched_count=${row?.matched_count ?? 0}`,
    };
  }

  // 2. Fetch requirements template.
  const template = await supabase
    .from("templates")
    .select("*")
    .eq("template_name", "02_requirements_request_v1")
    .eq("is_active", true)
    .single();

  // 3. Render template variables.
  const renderedMessage = renderTemplate(template.content, {
    propertyname: row.apt_address,
    address: row.apt_address,
    city: "",
    state: "",
    price: "",
    bedrooms: "",
    schedule: "",
    time: "",
    contactPhone: "",
    leadName: row.lead_name,
  });

  // 4. Send through existing Quo send/web auto-clicker logic.
  const sendResult = await sendViaQuo({
    phone,
    message: renderedMessage,
  });

  if (!sendResult.ok) {
    // mark human review or log error
    return {
      action: "send_failed",
      error: sendResult.error,
    };
  }

  // 5. Mark lead requirements sent.
  await supabase.rpc("mark_lead_requirements_sent", {
    p_lead_id: row.lead_id,
    p_inbound_sms: inboundText,
    p_outbound_sms: renderedMessage,
  });

  return {
    action: "requirements_sent",
    lead_id: row.lead_id,
  };
}
